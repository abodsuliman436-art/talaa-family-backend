const express = require("express"); const cors = require("cors"); const bcrypt = require("bcryptjs"); const jwt = require("jsonwebtoken"); const Database = require("better-sqlite3"); const app = express(); app.use(cors()); app.use(express.json()); const PORT = process.env.PORT || 3000; const JWT_SECRET = process.env.JWT_SECRET || "TALA_FAMILY_SECRET_CHANGE_ME"; const db = new Database("family.db"); db.pragma("journal_mode = WAL"); } // إضافة بيانات الأبناء if (!columnExists("users", "children_data")) { db.prepare(` ALTER TABLE users ADD COLUMN children_data TEXT DEFAULT '[]' `).run(); } // =============================== // إنشاء المجالس الأساسية // =============================== const createCouncil = db.prepare(` INSERT OR IGNORE INTO councils (name) VALUES (?) `); createCouncil.run( "مجلس العائلة في مخيم المغازي" ); createCouncil.run( "مجلس العائلة في مخيم النصيرات" ); // =============================== // دوال التحقق // =============================== function isValidName(name) { if (typeof name !== "string") { return false; } const parts = name .trim() .split(/\s+/) .filter(Boolean); return parts.length === 3; } function isValidNationalId(id) { return /^\d{9}$/.test(String(id)); } function isValidPhone(phone) { return /^(059|056)\d{7}$/.test(String(phone)); } // =============================== // التحقق من التوكن // =============================== function authenticate(req, res, next) { const authHeader = req.headers.authorization; if (!authHeader) { return res.status(401).json({ success: false, message: "يجب تسجيل الدخول أولاً" }); } const parts = authHeader.split(" "); if ( parts.length !== 2 || parts[0] !== "Bearer" ) { return res.status(401).json({ success: false, message: "رمز الدخول غير صالح" }); } const token = parts[1]; try { const decoded = jwt.verify( token, JWT_SECRET ); req.user = decoded; next(); } catch (error) { return res.status(401).json({ success: false, message: "انتهت صلاحية تسجيل الدخول" }); } }// =============================== // تسجيل حساب جديد // =============================== app.post("/api/register", async (req, res) => { try { const { name, nationalId, phone, password, maritalStatus, wivesCount, wives, childrenCount, children } = req.body; } catch (error) { console.error( "REGISTER ERROR:", error ); return res.status(500).json({ success: false, message: "حدث خطأ أثناء إنشاء الحساب" }); } });// =============================== // تسجيل الدخول // =============================== app.post("/api/login", async (req, res) => { try { const { identifier, password } = req.body; // =============================== // التحقق من البيانات // =============================== if (!identifier || !password) { return res.status(400).json({ success: false, message: "يرجى إدخال الاسم أو رقم الهوية أو رقم الهاتف وكلمة المرور" });
